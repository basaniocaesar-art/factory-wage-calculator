Cloud Sync Setup

Connect to Google Sheets so your data saves automatically and works from any device.

1. Open Google Sheets → create a blank sheet
2. Go to Extensions → Apps Script
3. Delete any code, paste the script shown below, then Save
4. Click Deploy → New deployment
5. Type = Web app, Access = Anyone, click Deploy
6. Copy the Web app URL and paste below
Show Apps Script code to copy 
function doGet() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName('Data') || ss.insertSheet('Data');
  var val = sheet.getRange('A1').getValue();
  return ContentService
    .createTextOutput(val || '{}')
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName('Data') || ss.insertSheet('Data');
  sheet.getRange('A1').setValue(e.postData.contents);
  return ContentService
    .createTextOutput('ok')
    .setMimeType(ContentService.MimeType.TEXT);
}
